name: Monitor Trading Bot Heartbeat

on:
  schedule:
    - cron: '15 * * * *'  # Runs automatically at minute 15 of every hour
  workflow_dispatch:      # Allows you to test it manually with a button click

jobs:
  check-bot-health:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Code Repository
        uses: actions/checkout@v4

      - name: Install Python Environment
        uses: actions/setup-python@v5
        with:
          python-version: '3.12'

      - name: Install Twilio Engine
        run: pip install twilio requests

      - name: Execute Health Assessment
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          TWILIO_SID: ${{ secrets.TWILIO_SID }}
          TWILIO_TOKEN: ${{ secrets.TWILIO_TOKEN }}
          MY_WHATSAPP_NUMBER: ${{ secrets.MY_WHATSAPP_NUMBER }}
          REPO_OWNER: ${{ github.repository_owner }}
          REPO_NAME: ${{ github.event.repository.name }}
        run: |
          python - << 'EOF'
          import os
          import requests
          from twilio.rest import Client

          # Pull security configurations natively from actions vault environments
          token = os.getenv('GITHUB_TOKEN')
          owner = os.getenv('REPO_OWNER')
          repo = os.getenv('REPO_NAME')
          
          headers = {
              "Authorization": f"Bearer {token}",
              "Accept": "application/vnd.github+json"
          }
          
          # Request the direct API tracking map for active execution runs
          url = f"https://github.com{owner}/{repo}/actions/workflows/run.yml/runs?status=in_progress"
          
          try:
              response = requests.get(url, headers=headers).json()
              active_runs = response.get('total_count', 0)
              
              print(f"[STATUS CHECK] Active running strategy instances detected: {active_runs}")
              
              # If active run instances drop to 0, the trading bot loop has failed
              if active_runs == 0:
                  error_msg = "🚨 CRITICAL WARNING: Your Delta Exchange India Trading Bot is currently STOPPED or has crashed in the cloud environment. Relaunching container pipelines immediately..."
                  print(error_msg)
                  
                  # 1. Trigger live phone notifications via Twilio channels
                  sid = os.getenv('TWILIO_SID')
                  auth = os.getenv('TWILIO_TOKEN')
                  to_num = os.getenv('MY_WHATSAPP_NUMBER')
                  
                  if sid and auth and to_num:
                      client = Client(sid, auth)
                      client.messages.create(
                          body=error_msg,
                          from_='whatsapp:+14155238886',
                          to=to_num
                      )
                      print("[WHATSAPP] Emergency failure alert transmitted successfully.")
                  
                  # 2. Force-trigger a fresh container to restart the trading bot automatically
                  trigger_url = f"https://github.com{owner}/{repo}/actions/workflows/run.yml/dispatches"
                  payload = {"ref": "main"}
                  requests.post(trigger_url, headers=headers, json=payload)
                  print("[RESTART Engine] Dispatched a fresh container pipeline successfully.")
              else:
                  print("[HEALTH CHECK] System is running normally. No faults detected.")
                  
          except Exception as e:
              print(f"[MONITOR FAULT] Failed to run health check operations: {e}")
          EOF
